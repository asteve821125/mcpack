async function showAdminLandList(player, page = 0, lands = undefined, title = "全服地皮") {
  loadData();
  const src = (lands ?? [...state.lands]).sort((a,b)=>a.id-b.id);
  const pageSize = 20;
  const pages = Math.max(1, Math.ceil(src.length / pageSize));
  page = Math.max(0, Math.min(page, pages - 1));
  const start = page * pageSize;
  const shown = src.slice(start, start + pageSize);
  const form = new ActionFormData().title(`${title} ${page+1}/${pages}`).body(src.length ? `共 ${src.length} 块，点击查看管理员操作` : "没有匹配的地皮。");
  for (const l of shown) {
    const z = landSize(l);
    form.button(`#${l.id} ${l.name}\n${l.owner} · ${z.width}×${z.length}`);
  }
  if (page > 0) form.button("上一页");
  if (page < pages - 1) form.button("下一页");
  form.button("返回管理员面板");
  const r = await showFormSafe(player, form, "管理员地皮列表");
  if (!r || r.canceled) return;
  if (r.selection < shown.length) return showAdminLandDetail(player, shown[r.selection], { page, lands: src, title });
  let idx = shown.length;
  if (page > 0) { if (r.selection === idx) return showAdminLandList(player, page - 1, src, title); idx++; }
  if (page < pages - 1) { if (r.selection === idx) return showAdminLandList(player, page + 1, src, title); idx++; }
  return showAdminMenu(player);
}

async function showAdminSearch(player) {
  const form = new ModalFormData().title("管理员搜索地皮").textField("地主名 / 地皮名称 / 地皮ID", "例如 Steve、商店、12");
  const r = await showFormSafe(player, form, "管理员搜索");
  if (!r || r.canceled) return showAdminMenu(player);
  const raw = String(r.formValues?.[0] ?? "").trim();
  if (!raw) return showAdminMenu(player);
  let results;
  if (/^#?\d+$/.test(raw)) {
    const id = Number(raw.replace("#", ""));
    const land = getLandById(id);
    results = land ? [land] : [];
  } else {
    const q = playerKey(raw);
    results = state.lands.filter(l => playerKey(l.owner).includes(q) || String(l.name).toLowerCase().includes(q));
  }
  return showAdminLandList(player, 0, results, `搜索：${raw}`);
}

  if (!loaded) return;
  const id = event.itemStack?.typeId ?? "";
  const fluidItem = id === "minecraft:water_bucket" || id === "minecraft:lava_bucket" || id === "minecraft:bucket" || id === "minecraft:powder_snow_bucket" || id === "minecraft:cod_bucket" || id === "minecraft:salmon_bucket" || id === "minecraft:pufferfish_bucket" || id === "minecraft:tropical_fish_bucket" || id === "minecraft:axolotl_bucket" || id === "minecraft:tadpole_bucket";
  const fireItem = id === "minecraft:flint_and_steel" || id === "minecraft:fire_charge";
  if (!fluidItem && !fireItem) return;
  try {
    const hit = event.source.getBlockFromViewDirection({ maxDistance: 7, includeLiquidBlocks: true, includePassableBlocks: true });
    if (!hit?.block) return;
    const land = getLandAt(hit.block.location, event.source.dimension.id);
    if (!land) return;
    const flag = fluidItem ? "fluid" : "fire";
    if (!can(event.source, land, flag)) {
      event.cancel = true;
      deny(event.source, land, fluidItem ? "使用流体桶" : "点火");
    }
  } catch (e) { logError("桶/点火保护失败", e); }
});

// 稳定 2.9.0 只有 pistonActivate after-event：用于检测跨地皮活塞并阻止后续持续动作。
// 无法在稳定 API 中真正 pre-cancel 第一次红石推动，因此这里只做边界检测与告警/停用活塞兜底。
if (world.afterEvents.pistonActivate) world.afterEvents.pistonActivate.subscribe((event) => {
  if (!loaded || !event.isExpanding) return;
  try {
    const pistonBlock = event.piston.block;
    const dim = pistonBlock.dimension;
    const pistonLand = getLandAt(pistonBlock.location, dim.id);
    const attached = event.piston.getAttachedBlocksLocations();
    let crossedProtected = false;
    for (const loc of attached) {
      const targetLand = getLandAt(loc, dim.id);
      if (targetLand && targetLand !== pistonLand && !targetLand.flags.piston) { crossedProtected = true; break; }
      if (pistonLand && !pistonLand.flags.piston && !pointInLand(loc, dim.id, pistonLand)) { crossedProtected = true; break; }
    }
    if (crossedProtected) {
      system.run(() => {
        try {
          // 稳定 API 无可取消的活塞前置事件；检测到违规跨界后移除活塞，避免连续推动。
          pistonBlock.setType("minecraft:air");
          console.warn(`[myLand] 检测并移除跨地皮活塞 @ ${pistonBlock.location.x},${pistonBlock.location.y},${pistonBlock.location.z}`);
        } catch (e) { logError("跨地皮活塞兜底失败", e); }
      });
    }
  } catch (e) { logError("活塞边界检测失败", e); }
});

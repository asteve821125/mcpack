function createLandFromSelection(player, second) {
  loadData();
  const key = playerKey(player.name);
  const first = selections.get(key);
  if (!first) {
    selections.set(key, second);
    tell(player, `已选择第一点：§e${second.x} ${second.y} ${second.z}§r。再用木锄选择第二点。`);
    return;
  }
  if (first.dimension !== second.dimension) {
    selections.set(key, second);
    tell(player, "两个点必须在同一维度；当前点已重新设为第一点。");
    return;
  }
  const candidateId = getNextFreeLandId();
  const land = normalizeLand({
    id: candidateId, owner: player.name, name: `地皮 #${candidateId}`, dimension: first.dimension,
    x1: first.x, z1: first.z, x2: second.x, z2: second.z,
    createdAt: Date.now(), trust: [], blacklist: [], flags: DEFAULT_FLAGS,
  }, candidateId);
  const size = landSize(land);
  if ((size.width < MIN_SIDE || size.length < MIN_SIDE) && !isBypass(player)) {
    tell(player, `地皮边长至少 §e${MIN_SIDE}×${MIN_SIDE}§r；当前为 ${size.width}×${size.length}。请重新选择第二点。`);
    return;
  }
  if ((size.width > MAX_WIDTH || size.length > MAX_LENGTH) && !isBypass(player)) {
    tell(player, `地皮最大 §e${MAX_WIDTH}×${MAX_LENGTH}§r；当前为 ${size.width}×${size.length}。请重新选择第二点。`);
    return;
  }
  const conflict = state.lands.find(l => overlaps(l, land));
  if (conflict) {
    tell(player, `划定失败：与 §e#${conflict.id} ${conflict.name}§r（${conflict.owner}）重叠。`);
    return;
  }
  pendingClaims.set(key, land);
  selections.delete(key);
  system.run(() => { void showClaimConfirm(player, land); });
}

function blockIsSafeAir(block) {
  try { return !!block && block.isAir === true; } catch { return false; }
}
function isSafeStand(dimension, x, y, z) {
  try {
    const feet = dimension.getBlock({ x, y, z });
    const head = dimension.getBlock({ x, y: y + 1, z });
    const below = dimension.getBlock({ x, y: y - 1, z });
    if (!feet || !head || !below) return false;
    return blockIsSafeAir(feet) && blockIsSafeAir(head) && !below.isAir && !below.isLiquid;
  } catch { return false; }
}
function findSafeNear(dimension, desired, land, radius = 5) {
  const cx = Math.floor(desired.x), cz = Math.floor(desired.z);
  const baseY = Math.floor(desired.y);
  for (let r = 0; r <= radius; r++) {
    for (let dx = -r; dx <= r; dx++) for (let dz = -r; dz <= r; dz++) {
      if (r > 0 && Math.abs(dx) !== r && Math.abs(dz) !== r) continue;
      const x = cx + dx, z = cz + dz;
      if (land && (x < land.x1 || x > land.x2 || z < land.z1 || z > land.z2)) continue;
      for (const dy of [0,1,2,3,-1,-2,-3,-4,-5,4,5,6,-6,-7,-8]) {
        const y = baseY + dy;
        if (isSafeStand(dimension, x, y, z)) return { x: x + 0.5, y, z: z + 0.5 };
      }
    }
  }
  return undefined;
}
function computeSafeTarget(land) {
  const dimension = world.getDimension(land.dimension.replace("minecraft:", ""));
  if (land.home) {
    const safeHome = findSafeNear(dimension, land.home, land, 4);
    if (safeHome) return { dimension, location: safeHome };
  }
  const x = Math.floor((land.x1 + land.x2) / 2), z = Math.floor((land.z1 + land.z2) / 2);
  try {
    const top = dimension.getTopmostBlock({ x, z });
    if (top) {
      const safe = findSafeNear(dimension, { x, y: top.location.y + 1, z }, land, 6);
      if (safe) return { dimension, location: safe };
    }
  } catch {}
  const safe = findSafeNear(dimension, { x, y: 80, z }, land, 6);
  return safe ? { dimension, location: safe } : undefined;
}
function safeTeleport(player, land, options = {}) {
  try {
    const now = Date.now();
    const key = playerKey(player.name);
    const privileged = options.admin === true || isBypass(player);
    if (!privileged && isBlacklisted(player, land)) return tell(player, "你已被该地皮加入黑名单，无法传送进入。");
    if (!options.ignoreCooldown && !privileged) {
      const combat = combatUntil.get(key) ?? 0;
      if (combat > now) return tell(player, `战斗状态中，${Math.ceil((combat-now)/1000)} 秒后才能使用地皮传送。`);
      const cd = teleportCooldownUntil.get(key) ?? 0;
      if (cd > now) return tell(player, `传送冷却中，还需 ${Math.ceil((cd-now)/1000)} 秒。`);
    }
    const target = computeSafeTarget(land);
    if (!target) return tell(player, "没有找到安全落点。请地主重新设置 Home，或清理传送点周围两格空间。");
    player.teleport(target.location, { dimension: target.dimension, checkForBlocks: true, forceProvidedPositionOnDimensionChange: true });
    if (!privileged) teleportCooldownUntil.set(key, now + TELEPORT_COOLDOWN_MS);
    tell(player, `已安全传送到 §e#${land.id} ${land.name}§r。`);
  } catch (e) { logError("地皮传送失败", e); tell(player, "传送失败：目标区块可能未加载或没有安全落点。"); }
}

function borderPoints(land) {
  const pts = [];
  const width = land.x2 - land.x1, length = land.z2 - land.z1;
  const perimeter = Math.max(1, 2 * (width + length));
  const count = Math.min(BORDER_MAX_POINTS, Math.max(8, Math.ceil(perimeter / 8)));
  for (let i = 0; i < count; i++) {
    const t = i / count * perimeter;
    let x, z;
    if (t <= width) { x = land.x1 + t; z = land.z1; }
    else if (t <= width + length) { x = land.x2; z = land.z1 + (t-width); }
    else if (t <= 2*width + length) { x = land.x2 - (t-width-length); z = land.z2; }
    else { x = land.x1; z = land.z2 - (t-2*width-length); }
    pts.push({ x: x + 0.5, z: z + 0.5 });
  }
  return pts;
}

  tell(player, "已显示低频地皮边界约 10 秒；粒子仅你自己可见。再次点击可刷新时长。");
}

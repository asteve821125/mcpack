function copyPlayerLocation(player){return {x:player.location.x,y:player.location.y,z:player.location.z,dimension:player.dimension.id};}
function ejectBlacklisted(player, land){
  try{
    const key=playerKey(player.name), prev=lastSafeLocation.get(key);
    if(prev && prev.dimension===player.dimension.id){player.teleport({x:prev.x,y:prev.y,z:prev.z},{dimension:player.dimension,checkForBlocks:true});tell(player,`§c你已被 #${land.id} ${land.name} 加入黑名单，无法进入。§r`);return;}
    const dim=player.dimension,candidates=[{x:land.x1-3,z:Math.floor((land.z1+land.z2)/2)},{x:land.x2+3,z:Math.floor((land.z1+land.z2)/2)},{x:Math.floor((land.x1+land.x2)/2),z:land.z1-3},{x:Math.floor((land.x1+land.x2)/2),z:land.z2+3}];
    for(const c of candidates){try{const top=dim.getTopmostBlock(c);if(!top)continue;const loc=findSafeNear(dim,{x:c.x,y:top.location.y+1,z:c.z},undefined,2);if(loc){player.teleport(loc,{dimension:dim,checkForBlocks:true});tell(player,`§c你已被 #${land.id} ${land.name} 加入黑名单，无法进入。§r`);return;}}catch{}}
    tell(player,"你在该地皮黑名单中，请立即离开此区域。");
  }catch(e){logError("黑名单驱离失败",e);}
}

        continue;
      }

        player.removeTag(BRIDGE_TAG_MENU);

system.runInterval(() => {
  if (!loaded) return;
  for (const player of world.getAllPlayers()) {
    const pkey=playerKey(player.name);
    const land = getLandAt(player.location, player.dimension.id);
    if(land && isBlacklisted(player,land)){
      ejectBlacklisted(player,land);
      lastShownLand.set(pkey,"");
      continue;
    }
    lastSafeLocation.set(pkey,copyPlayerLocation(player));
    const key = land ? String(land.id) : "";
    const previous = lastShownLand.get(pkey) ?? "";
    if (key !== previous) {
      const oldLand=previous?getLandById(Number(previous)):undefined;
      lastShownLand.set(pkey, key);
      try {
        if(oldLand?.farewell) tell(player,`§7离开 ${oldLand.name}：§r${oldLand.farewell}`);
        if (land) {
          player.onScreenDisplay.setActionBar(`§e${land.name}§r §7(#${land.id}) | ${land.owner}§r`);
          if(land.welcome) tell(player,`§a${land.name}：§r${land.welcome}`);
        } else if (previous) player.onScreenDisplay.setActionBar("§7已离开地皮§r");
      } catch (e) { logError("进出地皮提示", e); }
    }
  }
}, 10);

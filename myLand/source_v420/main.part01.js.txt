function getBridgeObjective() {
  try {
    return world.scoreboard.getObjective(BRIDGE_OBJECTIVE) ?? world.scoreboard.addObjective(BRIDGE_OBJECTIVE, "插件菜单协同");
  } catch {
    return undefined;
  }
}
function pulseBridge() {
  const obj = getBridgeObjective();
  if (!obj) return;
  try {
    obj.setScore(BRIDGE_SELF, BRIDGE_TTL);
    for (const peer of BRIDGE_PEERS) {
      const score = obj.getScore(peer) ?? 0;
      if (score > 0) obj.setScore(peer, Math.max(0, score - 1));
    }
  } catch {}
}
function bridgeAlive(name) {
  const obj = getBridgeObjective();
  if (!obj) return false;
  try { return (obj.getScore(name) ?? 0) > 0; } catch { return false; }
}

function floor(n) { return Math.floor(Number(n)); }
function cleanName(value, fallback = "未命名地皮") {
  const text = String(value ?? "").trim().replace(/[\r\n§]/g, "");
  return (text || fallback).slice(0, 32);
}
function cleanText(value, max = 120, fallback = "") {
  const text = String(value ?? "").trim().replace(/[\r\n§]/g, " ").replace(/\s+/g, " ");
  return (text || fallback).slice(0, max);
}
function uniqueNames(list) {
  const out = [];
  for (const v of Array.isArray(list) ? list : []) {
    const n = cleanText(v, 40);
    if (n && !out.some(x => nameEquals(x, n))) out.push(n);
  }
  return out;
}
function playerKey(name) { return String(name ?? "").toLowerCase(); }
function nameEquals(a, b) { return playerKey(a) === playerKey(b); }
function cloneFlags(raw) {
  const result = { ...DEFAULT_FLAGS };
  if (raw && typeof raw === "object") {
    for (const key of FLAG_NAMES) if (typeof raw[key] === "boolean") result[key] = raw[key];
  }
  return result;
}

function normalizeLand(raw, fallbackId = 0) {
  if (!raw || typeof raw !== "object") return undefined;
  const x1 = Number(raw.x1), z1 = Number(raw.z1), x2 = Number(raw.x2), z2 = Number(raw.z2);
  if (![x1, z1, x2, z2].every(Number.isFinite)) return undefined;
  const owner = typeof raw.owner === "string" ? raw.owner : "";
  if (!owner) return undefined;
  const id = Number.isInteger(raw.id) && raw.id > 0 ? raw.id : fallbackId;
  const trust = Array.isArray(raw.trust) ? [...new Set(raw.trust.filter(v => typeof v === "string" && v.trim()).map(v => v.trim()))] : [];
  return {
    id,
    owner,
    name: cleanName(raw.name, `地皮 #${id || "?"}`),
    dimension: typeof raw.dimension === "string" ? raw.dimension : "minecraft:overworld",
    x1: floor(Math.min(x1, x2)), z1: floor(Math.min(z1, z2)),
    x2: floor(Math.max(x1, x2)), z2: floor(Math.max(z1, z2)),
    createdAt: Number.isFinite(raw.createdAt) ? raw.createdAt : Date.now(),
    trust,
    blacklist: uniqueNames(raw.blacklist).filter(n => !nameEquals(n, owner)),
    flags: cloneFlags(raw.flags),
    description: cleanText(raw.description, 160),
    welcome: cleanText(raw.welcome, 100),
    farewell: cleanText(raw.farewell, 100),
    publicListed: raw.publicListed === true,
    forSale: raw.forSale === true && Number.isFinite(Number(raw.salePrice)) && Number(raw.salePrice) > 0,
    salePrice: Math.max(0, Math.floor(Number(raw.salePrice) || 0)),
    home: raw.home && [raw.home.x, raw.home.y, raw.home.z].every(Number.isFinite)
      ? { x: Number(raw.home.x), y: Number(raw.home.y), z: Number(raw.home.z) }
      : undefined,
  };
}

function getNextFreeLandId(lands = state.lands) {
  const used = new Set((lands ?? []).map(l => Number(l?.id)).filter(id => Number.isInteger(id) && id > 0));
  let id = 1;
  while (used.has(id)) id++;
  return id;
}
function refreshNextId() {
  state.nextId = getNextFreeLandId(state.lands);
  return state.nextId;
}

function serializeState() { return JSON.stringify(state); }
function emptyState() { return { version: SCHEMA_VERSION, nextId: 1, lands: [], economy: { balances: {} } }; }
function parseState(text) {
  const parsed = JSON.parse(text);
  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) throw new Error("invalid state root");
  if (parsed.version !== SCHEMA_VERSION) throw new Error(`unsupported schema ${parsed.version}`);
  if (!Array.isArray(parsed.lands)) throw new Error("invalid lands array");
  if (!parsed.economy || typeof parsed.economy !== "object" || !parsed.economy.balances || typeof parsed.economy.balances !== "object") {
    throw new Error("invalid economy data");
  }
  const lands = [];
  const used = new Set();
  for (const raw of parsed.lands) {
    const land = normalizeLand(raw);
    if (!land || !Number.isInteger(land.id) || land.id <= 0 || used.has(land.id)) throw new Error("invalid or duplicate land id");
    used.add(land.id);
    lands.push(land);
  }
  const balances = {};
  for (const [key, value] of Object.entries(parsed.economy.balances)) {
    const amount = Math.floor(Number(value));
    if (key && Number.isFinite(amount) && amount >= 0) balances[String(key).toLowerCase()] = amount;
  }
  return { version: SCHEMA_VERSION, nextId: getNextFreeLandId(lands), lands, economy: { balances } };
}

function loadData() {
  if (loaded) return;
  loaded = true;
  try {
    const metaRaw = world.getDynamicProperty(META_KEY);
    if (typeof metaRaw !== "string" || !metaRaw) {
      state = emptyState();
      return;
    }
    const meta = JSON.parse(metaRaw);
    if (meta.version !== SCHEMA_VERSION) throw new Error(`unsupported meta schema ${meta.version}`);
    const count = Number(meta.chunks);
    if (!Number.isInteger(count) || count < 1) throw new Error("invalid chunk count");
    let text = "";
    for (let i = 0; i < count; i++) {
      const part = world.getDynamicProperty(`${CHUNK_KEY}${i}`);
      if (typeof part !== "string") throw new Error(`missing data chunk ${i}`);
      text += part;
    }
    state = parseState(text);

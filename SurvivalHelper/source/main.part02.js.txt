      const feet = player.dimension.getBlock({ x, y: y + 1, z });
      const head = player.dimension.getBlock({ x, y: y + 2, z });
      if (blockIsFloor(floor) && blockIsPassable(feet) && blockIsPassable(head)) {
        player.teleport({ x: x + 0.5, y: y + 1.01, z: z + 0.5 }, { dimension: player.dimension, checkForBlocks: false });
        player.sendMessage("§b[生存辅助] §f已通过潜行电梯向上移动。");
        return;
      }
    } catch {}
  }
  player.sendMessage("§e[生存辅助] 上方 32 格内没有找到可站立的平台。");
}

function roundCoord(value) {
  return Math.floor(value);
}

function dimensionName(id) {
  if (id === "minecraft:overworld" || id === "overworld") return "主世界";
  if (id === "minecraft:nether" || id === "nether") return "下界";
  if (id === "minecraft:the_end" || id === "the_end") return "末地";
  return id;
}

function getDimensionBySavedId(id) {
  if (id === "minecraft:overworld" || id === "overworld") return world.getDimension("overworld");
  if (id === "minecraft:nether" || id === "nether") return world.getDimension("nether");
  if (id === "minecraft:the_end" || id === "the_end") return world.getDimension("the_end");
  return undefined;
}

function makeLocationRecord(player) {
  return {
    x: player.location.x,
    y: player.location.y,
    z: player.location.z,
    dimension: player.dimension.id
  };
}

function parseLocationRecord(raw) {
  if (!raw) return undefined;
  try {
    const data = JSON.parse(raw);
    if (
      typeof data?.x !== "number" ||
      typeof data?.y !== "number" ||
      typeof data?.z !== "number" ||
      typeof data?.dimension !== "string"
    ) return undefined;
    return data;
  } catch {
    return undefined;
  }
}

function validHomeRecord(home) {
  return !!home &&
    typeof home.name === "string" && home.name.trim().length > 0 &&
    typeof home.x === "number" && typeof home.y === "number" &&
    typeof home.z === "number" && typeof home.dimension === "string";
}

function normalizeHomeName(name, fallback = "传送点") {
  const text = String(name ?? "").trim().replace(/[\r\n§]/g, " ").slice(0, 24);
  return text || fallback;
}

function getPersonalHomeLimit(player) {
  try {
    const value = player.getDynamicProperty(PROP_HOME_LIMIT);
    if (typeof value === "number" && Number.isFinite(value)) {
      return Math.max(0, Math.min(MAX_STORED_PERSONAL_HOMES, Math.floor(value)));
    }
  } catch {}
  return DEFAULT_PERSONAL_HOME_LIMIT;
}

function setPersonalHomeLimit(player, limit) {
  const value = Math.max(0, Math.min(MAX_STORED_PERSONAL_HOMES, Math.floor(Number(limit) || 0)));
  try {
    player.setDynamicProperty(PROP_HOME_LIMIT, value);
    return true;
  } catch (error) {
    console.warn(`[生存辅助] 保存传送点上限失败: ${error}`);
    return false;
  }
}

function getPersonalHomes(player) {
  let homes = [];
  const raw = getStringProperty(player, PROP_HOMES);
  if (raw) {
    try {
      const data = JSON.parse(raw);
      if (Array.isArray(data)) homes = data.filter(validHomeRecord).slice(0, MAX_STORED_PERSONAL_HOMES);
    } catch (error) {
      console.warn(`[生存辅助] 读取个人传送点失败: ${error}`);
    }
  }

  // One-time migration from v2.5.x single home to the v2.6.0 multi-home list.
  if (homes.length === 0) {
    const legacy = parseLocationRecord(getStringProperty(player, PROP_HOME));
    if (legacy) {
      homes = [{ name: "默认家园", ...legacy }];
      savePersonalHomes(player, homes);
      clearProperty(player, PROP_HOME);
      try { player.sendMessage("§a[生存辅助] §f旧版家园已自动迁移为“默认家园”。"); } catch {}
    }
  }
  return homes;
}

function savePersonalHomes(player, homes) {
  const clean = (Array.isArray(homes) ? homes : []).filter(validHomeRecord).slice(0, MAX_STORED_PERSONAL_HOMES);
  return setStringProperty(player, PROP_HOMES, JSON.stringify(clean));
}

function makeNamedHome(player, name) {
  return { name: normalizeHomeName(name), ...makeLocationRecord(player) };
}

function formatLocation(data) {
  return `${dimensionName(data.dimension)}  X:${roundCoord(data.x)} Y:${roundCoord(data.y)} Z:${roundCoord(data.z)}`;
}

function teleportToRecord(player, data, label) {
  const dimension = getDimensionBySavedId(data.dimension);
  if (!dimension) {
    player.sendMessage(`§c[生存辅助] ${label}所在维度无法识别。`);
    return;
  }

  try {
    player.teleport(
      { x: data.x, y: data.y, z: data.z },
      { dimension, checkForBlocks: false }
    );
    player.sendMessage(`§a[生存辅助] §f已传送到${label}：${formatLocation(data)}`);
  } catch (error) {
    player.sendMessage(`§c[生存辅助] 传送失败，请稍后重试。`);
    console.warn(`[生存辅助] ${label}传送失败: ${error}`);
  }
}

function isHoldingTnt(player) {
  try {
    const inventory = player.getComponent("minecraft:inventory")?.container;
    return inventory?.getItem(player.selectedSlotIndex)?.typeId === TNT_ID;
  } catch {
    return false;
  }
}

function consumeOneTnt(player) {
  if (player.getGameMode() === GameMode.Creative) return;

  const inventory = player.getComponent("minecraft:inventory")?.container;
  if (!inventory) return;

  const slot = player.selectedSlotIndex;
  const stack = inventory.getItem(slot);
  if (!stack || stack.typeId !== TNT_ID) return;

  if (stack.amount <= 1) {
    inventory.setItem(slot, undefined);
  } else {
    stack.amount -= 1;
    inventory.setItem(slot, stack);
  }
}

function updateTntDimensionSession(player) {
  const current = player.dimension.id;
  const previous = lastKnownDimension.get(player.id);

  if (previous !== current) {
    // Every dimension transition starts a fresh TNT reminder state.
    // This makes re-entering the Nether require the one-time reminder again.
    netherTntAcknowledged.delete(player.id);
    lastKnownDimension.set(player.id, current);
  }

  return current;
}

function throwTnt(player) {
  if (!tntEnabledFor(player)) return;

  const dimensionId = updateTntDimensionSession(player);
  if (dimensionId !== NETHER_ID) {
    const now = system.currentTick;
    const warnUntil = tntDimensionWarnUntil.get(player.id) ?? 0;
    if (now >= warnUntil) {
      tntDimensionWarnUntil.set(player.id, now + 40);
      system.run(() => {
        try {
          player.sendMessage("§c[生存辅助] §fTNT 投掷仅允许在下界使用。");
        } catch {}
      });
    }
    return;
  }

  // Only the first TNT-use attempt of each Nether stay shows a reminder.
  // The reminder itself does not throw TNT; all later attempts in this same
  // Nether stay throw immediately. Leaving the Nether resets this state.
  if (!netherTntAcknowledged.has(player.id)) {
    netherTntAcknowledged.add(player.id);
    system.run(() => {
      try {
        player.sendMessage("§e[生存辅助] §fTNT 投掷已就绪：本次进入下界只提醒这一次；再次潜行使用 TNT 即可投掷，离开下界后会重置。");
      } catch {}
    });
    return;
  }

  const view = player.getViewDirection();
  const location = player.location;

  system.run(() => {
    try {
      if (!isHoldingTnt(player) || !tntEnabledFor(player)) return;
      if (player.dimension.id !== NETHER_ID) return;

      const tnt = player.dimension.spawnEntity(TNT_ID, {
        x: location.x,
        y: location.y + 1.2,
        z: location.z
      });

      tnt.applyImpulse({
        x: view.x * 0.5,
        y: view.y * 0.5 + 0.2,
        z: view.z * 0.5
      });

      consumeOneTnt(player);
    } catch (error) {
      console.warn(`[生存辅助] TNT 投掷失败: ${error}`);
    }
  });
}

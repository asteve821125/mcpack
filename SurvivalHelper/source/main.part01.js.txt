function getBridgeObjective() {
  try {
    return world.scoreboard.getObjective(BRIDGE_OBJECTIVE) ?? world.scoreboard.addObjective(BRIDGE_OBJECTIVE, "插件菜单协同");
  } catch {
    return undefined;
  }
}
function pulseBridge() {
  const obj = getBridgeObjective();
  if (!obj) return;
  try {
    obj.setScore(BRIDGE_SELF, BRIDGE_TTL);
    const other = obj.getScore(BRIDGE_OTHER) ?? 0;
    if (other > 0) obj.setScore(BRIDGE_OTHER, Math.max(0, other - 1));
  } catch {}
}
function bridgeAlive(name) {
  const obj = getBridgeObjective();
  if (!obj) return false;
  try { return (obj.getScore(name) ?? 0) > 0; } catch { return false; }
}

function commandPlayer(origin) {
  try {
    const entity = origin.sourceEntity;
    return entity?.typeId === "minecraft:player" ? entity : undefined;
  } catch {
    return undefined;
  }
}

function commandOk(message) {
  return { status: CustomCommandStatus.Success, message };
}

function commandFail(message) {
  return { status: CustomCommandStatus.Failure, message };
}

function boolProperty(target, key, defaultValue = true) {
  try {
    const value = target.getDynamicProperty(key);
    return typeof value === "boolean" ? value : defaultValue;
  } catch {
    return defaultValue;
  }
}

function setBoolProperty(target, key, value) {
  try {
    target.setDynamicProperty(key, value);
    return true;
  } catch (error) {
    console.warn(`[生存辅助] 保存设置失败 (${key}): ${error}`);
    return false;
  }
}

function getStringProperty(target, key) {
  try {
    const value = target.getDynamicProperty(key);
    return typeof value === "string" ? value : undefined;
  } catch {
    return undefined;
  }
}

function setStringProperty(target, key, value) {

}

function allowedFor(player, allowKey) {
  return !allowKey || boolProperty(player, allowKey, true);
}

function featureEnabled(player, globalKey, personalKey, allowKey) {
  if (!boolProperty(world, globalKey, true)) return false;
  if (!allowedFor(player, allowKey)) return false;
  return personalKey ? boolProperty(player, personalKey, true) : true;
}

const CHAIN_BATCH_SIZE = 4;

function sameChainFamily(startId, candidateId, kind) {
  if (kind === "ore") return ORE_FAMILY.get(candidateId) === ORE_FAMILY.get(startId);
  if (kind === "log") return candidateId === startId;
  return false;
}

function getHeldItem(player) {
  try {
    const inventory = player.getComponent("minecraft:inventory");
    const container = inventory?.container;
    if (!container) return undefined;
    return container.getItem(player.selectedSlotIndex);
  } catch {
    return undefined;
  }
}

function hasCorrectChainTool(player, kind) {
  const item = getHeldItem(player);
  if (!item) return false;
  return kind === "ore" ? PICKAXE_IDS.has(item.typeId) : AXE_IDS.has(item.typeId);
}

function getUnbreakingLevel(item) {
  try {
    const enchantable = item.getComponent("minecraft:enchantable");
    if (!enchantable) return 0;
    return enchantable.getEnchantment("unbreaking")?.level ?? enchantable.getEnchantment("minecraft:unbreaking")?.level ?? 0;
  } catch {
    return 0;
  }
}

function damageHeldTool(player, expectedTypeId, expectedSlot) {
  try {
    if (player.getGameMode() === GameMode.Creative) return true;
    if (player.selectedSlotIndex !== expectedSlot) return false;

    const inventory = player.getComponent("minecraft:inventory");
    const container = inventory?.container;
    if (!container) return false;
    const item = container.getItem(expectedSlot);
    if (!item || item.typeId !== expectedTypeId) return false;

    const durability = item.getComponent("minecraft:durability");
    if (!durability || durability.unbreakable) return true;

    const unbreaking = Math.max(0, Math.min(3, getUnbreakingLevel(item)));
    const damageChance = durability.getDamageChance(unbreaking);
    if (Math.random() * 100 >= damageChance) return true;

    if (durability.damage + 1 >= durability.maxDurability) {
      container.setItem(expectedSlot, undefined);
      try { player.playSound("random.break"); } catch {}
      return false;
    }

    durability.damage += 1;
    container.setItem(expectedSlot, item);
    return true;
  } catch {
    return false;
  }
}

function collectChainBlocks(dimension, origin, startId, kind, maxBlocks) {
  const queue = [{ x: origin.x, y: origin.y, z: origin.z }];
  const seen = new Set([`${origin.x},${origin.y},${origin.z}`]);
  const result = [];

  while (queue.length && result.length < maxBlocks) {
    const current = queue.shift();
    for (let dx = -1; dx <= 1; dx++) {
      for (let dy = -1; dy <= 1; dy++) {
        for (let dz = -1; dz <= 1; dz++) {
          if (dx === 0 && dy === 0 && dz === 0) continue;
          const loc = { x: current.x + dx, y: current.y + dy, z: current.z + dz };
          const key = `${loc.x},${loc.y},${loc.z}`;
          if (seen.has(key)) continue;
          seen.add(key);

          // Keep tree felling local to the trunk and avoid chasing decorative structures too far.
          if (kind === "log") {
            if (loc.y < origin.y - 1) continue;
            if (Math.abs(loc.x - origin.x) > CHAIN_MAX_RADIUS || Math.abs(loc.z - origin.z) > CHAIN_MAX_RADIUS) continue;
          }

          try {
            const block = dimension.getBlock(loc);
            if (!block || !sameChainFamily(startId, block.typeId, kind)) continue;
            result.push(loc);
            queue.push(loc);
            if (result.length >= maxBlocks) return result;
          } catch {}
        }
      }
    }
  }
  return result;
}

function startChainBreak(player, origin, startId, kind, maxBlocks) {
  if (!hasCorrectChainTool(player, kind)) {
    player.sendMessage(kind === "ore" ? "§e[连锁挖矿] 请使用镐触发连锁挖矿。" : "§e[连锁砍树] 请使用斧触发连锁砍树。");
    return;
  }

  const dimension = player.dimension;
  const dimensionId = dimension.id;
  const held = getHeldItem(player);
  if (!held) return;
  const toolTypeId = held.typeId;
  const toolSlot = player.selectedSlotIndex;
  const targets = collectChainBlocks(dimension, origin, startId, kind, maxBlocks);
  if (targets.length === 0) return;

  let index = 0;
  let broken = 0;

  const processBatch = () => {
    if (!player?.isValid || player.dimension.id !== dimensionId || player.selectedSlotIndex !== toolSlot) {
      if (broken > 0) player?.sendMessage?.(`§7[生存辅助] 连锁已停止，共额外处理 ${broken} 个方块。`);
      return;
    }

    let processedThisTick = 0;
    while (index < targets.length && processedThisTick < CHAIN_BATCH_SIZE) {
      const loc = targets[index++];
      processedThisTick++;

      try {
        const currentTool = getHeldItem(player);
        if (!currentTool || currentTool.typeId !== toolTypeId) break;

        const block = dimension.getBlock(loc);
        if (!block || !sameChainFamily(startId, block.typeId, kind)) continue;

        // Run /loot as the triggering player so `mainhand` resolves to that player's actual tool.
        // Only remove the block after loot generation succeeds, preventing silent item loss.
        player.runCommand(`loot spawn ${loc.x + 0.5} ${loc.y + 0.5} ${loc.z + 0.5} mine ${loc.x} ${loc.y} ${loc.z} mainhand`);
        block.setType("minecraft:air");
        broken++;

        if (!damageHeldTool(player, toolTypeId, toolSlot)) {
          player.sendMessage(`§e[生存辅助] 工具已损坏，连锁自动停止。`);
          if (broken > 0) player.sendMessage(`§a[${kind === "ore" ? "连锁挖矿" : "连锁砍树"}] §f额外处理 ${broken} 个方块。`);
          return;
        }
      } catch (error) {
        console.warn(`[生存辅助] 连锁处理失败: ${error}`);
      }
    }

    if (index < targets.length) {
      system.run(processBatch);
    } else if (broken > 0) {
      player.sendMessage(`§a[${kind === "ore" ? "连锁挖矿" : "连锁砍树"}] §f额外处理 ${broken} 个方块。`);
    }
  };

  system.run(processBatch);
}

function blockIsPassable(block) {
  if (!block) return false;
  return block.typeId === "minecraft:air" || block.typeId === "minecraft:cave_air" || block.typeId === "minecraft:void_air";
}

function blockIsFloor(block) {
  return !!block && !blockIsPassable(block) && !block.isLiquid;
}

function tryElevatorUp(player) {
  if (!elevatorEnabledFor(player) || !player.isSneaking) return;
  const x = Math.floor(player.location.x);
  const z = Math.floor(player.location.z);
  const startY = Math.floor(player.location.y);

  for (let y = startY + 2; y <= startY + 32; y++) {
    try {
      const floor = player.dimension.getBlock({ x, y, z });

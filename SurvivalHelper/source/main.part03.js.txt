async function showIncomingTpa(player) {
  cleanupTpaRequests();
  const requests = getIncomingRequests(player.id)
    .map((request) => ({ ...request, player: findPlayerById(request.requesterId) }))
    .filter((request) => request.player);

  if (requests.length === 0) {
    player.sendMessage("§e[生存辅助] 当前没有有效的 TPA 请求。")
    system.run(() => showSocialMenu(player));
    return;
  }

  try {
    const form = new ActionFormData().title("TPA 请求").body("选择一个请求进行处理：");
    for (const request of requests) form.button(request.player.name);
    form.button("返回");

    const response = await form.show(player);
    if (response.canceled) return;
    if (response.selection === requests.length) return system.run(() => showSocialMenu(player));

    const request = requests[response.selection];
    system.run(() => confirmTpaRequest(player, request.player));
  } catch (error) {
    console.warn(`[生存辅助] TPA 请求菜单失败: ${error}`);
  }
}

async function confirmTpaRequest(target, requester) {
  if (!target?.isValid || !requester?.isValid) return;

  try {
    const response = await new MessageFormData()
      .title("处理 TPA 请求")
      .body(`${requester.name} 想传送到你身边。`)
      .button1("拒绝")
      .button2("接受")
      .show(target);

    removeTpaRequest(target.id, requester.id);
    if (response.canceled) return;

    if (response.selection === 1) {
      if (!boolProperty(world, PROP_GLOBAL_TPA, true)) {
        target.sendMessage("§c[生存辅助] TPA 已被管理员关闭。")
        requester.sendMessage("§c[生存辅助] TPA 已被管理员关闭。")
        return;
      }
      try {
        requester.teleport(target.location, { dimension: target.dimension, checkForBlocks: false });
        requester.sendMessage(`§a[生存辅助] §f${target.name} 接受了你的 TPA 请求。`);
        target.sendMessage(`§a[生存辅助] §f已接受 ${requester.name} 的 TPA 请求。`);
      } catch (error) {
        target.sendMessage("§c[生存辅助] TPA 传送失败。")
        requester.sendMessage("§c[生存辅助] TPA 传送失败。")
        console.warn(`[生存辅助] TPA 传送失败: ${error}`);
      }
    } else {
      requester.sendMessage(`§c[生存辅助] ${target.name} 拒绝了你的 TPA 请求。`);
      target.sendMessage(`§e[生存辅助] 已拒绝 ${requester.name} 的 TPA 请求。`);
    }
  } catch (error) {
    console.warn(`[生存辅助] TPA 确认菜单失败: ${error}`);
  }
}

function addTpaRequest(targetId, requesterId) {
  cleanupTpaRequests();
  let requests = tpaRequests.get(targetId);
  if (!requests) {
    requests = new Map();
    tpaRequests.set(targetId, requests);
  }
  requests.set(requesterId, system.currentTick + TPA_EXPIRE_TICKS);
}

function removeTpaRequest(targetId, requesterId) {
  const requests = tpaRequests.get(targetId);
  if (!requests) return;
  requests.delete(requesterId);
  if (requests.size === 0) tpaRequests.delete(targetId);
}

function getIncomingRequests(targetId) {
  cleanupTpaRequests();
  const requests = tpaRequests.get(targetId);
  if (!requests) return [];
  return [...requests.entries()].map(([requesterId, expiresAt]) => ({ requesterId, expiresAt }));
}

function cleanupTpaRequests() {
  const now = system.currentTick;
  for (const [targetId, requests] of tpaRequests) {
    for (const [requesterId, expiresAt] of requests) {
      if (expiresAt < now || !findPlayerById(requesterId)) requests.delete(requesterId);
    }
    if (requests.size === 0 || !findPlayerById(targetId)) tpaRequests.delete(targetId);
  }
}

function findPlayerById(id) {
  return world.getAllPlayers().find((player) => player.id === id);
}

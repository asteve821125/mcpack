function playerKey(name) { return String(name ?? "").trim().toLowerCase(); }
function sameName(a,b) { return playerKey(a) === playerKey(b); }
function isAdmin(player) { try { return player.playerPermissionLevel === PlayerPermissionLevel.Operator; } catch { return false; } }
function dimCode(id) { return id === "minecraft:nether" ? "n" : id === "minecraft:the_end" ? "e" : "o"; }
function codeDim(code) { return code === "n" ? "minecraft:nether" : code === "e" ? "minecraft:the_end" : "minecraft:overworld"; }
function locKey(dimensionId, loc) { return `${dimCode(dimensionId)}:${Math.floor(loc.x)}:${Math.floor(loc.y)}:${Math.floor(loc.z)}`; }
function copyLoc(loc) { return { x: Math.floor(loc.x), y: Math.floor(loc.y), z: Math.floor(loc.z) }; }
function cleanText(value, max=40, fallback="") { const t=String(value??"").trim().replace(/[\r\n§]/g," ").replace(/\s+/g," "); return (t||fallback).slice(0,max); }
function formatMoney(v) { return `${Math.max(0,Math.floor(Number(v)||0)).toLocaleString()} 金币`; }

function getBridgeObjective() { try { return world.scoreboard.getObjective(BRIDGE_OBJECTIVE) ?? world.scoreboard.addObjective(BRIDGE_OBJECTIVE,"插件菜单协同"); } catch { return undefined; } }
function bridgeAlive(name) { const o=getBridgeObjective(); if(!o)return false; try{return (o.getScore(name)??0)>0;}catch{return false;} }
function pulseBridge() {
  const o=getBridgeObjective(); if(!o)return;
  try {
    o.setScore(BRIDGE_SELF,BRIDGE_TTL);
    for(const peer of BRIDGE_PEERS){ const n=o.getScore(peer)??0; if(n>0)o.setScore(peer,Math.max(0,n-1)); }
  } catch {}
}

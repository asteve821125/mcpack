function executePurchase(player,shop,fingerprint,qty){
  shop=getShopById(shop.id);if(!shop)return tell(player,"商店已不存在。");const product=shop.products[fingerprint];if(!product||product.price<=0)return tell(player,"该商品已下架。");const container=shopContainer(shop);if(!container)return tell(player,"商店箱子当前未加载，请靠近后重试。");
  const matching=[];let stock=0;for(let i=0;i<container.size;i++){const item=container.getItem(i);if(!item||itemFingerprint(item)!==fingerprint)continue;matching.push({slot:i,item});stock+=item.amount;}if(stock<qty)return tell(player,`库存不足，当前仅剩 ${stock}。`);
  const total=product.price*qty;const buyerBalance=getBalance(player.name,true);if(buyerBalance<total)return tell(player,`余额不足，需要 ${formatMoney(total)}。`);
  const transfers=[];let remain=qty;
  try{
    for(const m of matching){if(remain<=0)break;const take=Math.min(remain,m.item.amount);const out=m.item;out.amount=take;transfers.push(out);const left=m.item.amount-take;if(left<=0)container.setItem(m.slot,undefined);else{const keep=m.item;keep.amount=left;container.setItem(m.slot,keep);}remain-=take;}
    if(remain>0)throw new Error("inventory changed");
    setBalance(player.name,buyerBalance-total);const tax=Math.floor(total*(state.settings.taxPercent/100));addBalance(shop.owner,total-tax);
    const inv=player.getComponent("minecraft:inventory")?.container;
    for(const stack of transfers){let leftover=inv?.addItem(stack);if(leftover){const entity=player.dimension.spawnItem(leftover,{x:player.location.x,y:player.location.y+0.5,z:player.location.z});try{entity.applyImpulse({x:(Math.random()-.5)*.18,y:.18,z:(Math.random()-.5)*.18});}catch{}}}
    product.lastStock=stock-qty;recordTrade({time:Date.now(),buyer:player.name,seller:shop.owner,shopId:shop.id,item:product.label,qty,total,tax});saveData();tell(player,`购买成功：${product.label} ×${qty}，支付 ${formatMoney(total)}。${product.lastStock===0?" §e该商品已售罄。§r":""}`);const seller=world.getAllPlayers().find(x=>sameName(x.name,shop.owner));if(seller)tell(seller,`${player.name} 购买了 ${product.label} ×${qty}，你获得 ${formatMoney(total-tax)}。`);
  }catch(e){log("交易执行失败",e);tell(player,"交易失败，未能安全完成。请检查箱子库存后重试。");}
}
